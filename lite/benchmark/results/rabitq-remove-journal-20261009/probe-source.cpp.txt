
#include "vsag/lite/index.h"
#include <cstdlib>
#include <cmath>
#include <ctime>
#include <iomanip>
#include <iostream>
#include <new>
#include <sstream>
static int fail_after=-1;
static bool failure_fired=false;
static int allocations_after_failure=0;
void* operator new(std::size_t size) {
    if (fail_after == 0) { fail_after=-1; failure_fired=true; throw std::bad_alloc(); }
    if (failure_fired) ++allocations_after_failure;
    if (fail_after > 0) --fail_after;
    if (auto* result=std::malloc(size ? size : 1)) return result;
    throw std::bad_alloc();
}
void operator delete(void* p) noexcept { std::free(p); }
void operator delete(void* p,std::size_t) noexcept { std::free(p); }
void* operator new[](std::size_t n) { return ::operator new(n); }
void operator delete[](void* p) noexcept { std::free(p); }
void operator delete[](void* p,std::size_t) noexcept { std::free(p); }
std::string save(const vsag::lite::Index& index) {
    std::ostringstream output;
    if(!index.Save(output)) throw std::runtime_error("save");
    return output.str();
}
int measure_update(int mode) {
    constexpr uint64_t dim=128, count=2000, repeats=100;
    auto created=vsag::lite::Index::Create(dim);
    if(!created) return 21;
    std::vector<float> vector(dim);
    for(uint64_t d=0;d<dim;++d) vector[d]=std::sin(float(d)*0.031F);
    for(uint64_t row=0;row<count;++row) {
        vector[0]=float(row)*0.125F;
        if(!(*created)->Add(int64_t(row),vector.data(),dim)) return 22;
    }
    if(!(*created)->BuildGraph(vsag::lite::VectorStorage::RABITQ8,8,64)) return 23;
    vector[0]=0;
    const auto before=save(**created);
    const auto start=std::clock();
    for(uint64_t i=0;i<repeats;++i) {
        if(mode==3) {
            if(!(*created)->Remove(int64_t(i))) return 24;
        } else if(mode==2) {
            vector[0]=float(count+i)*0.125F;
            if(!(*created)->Add(int64_t(count+i),vector.data(),dim)) return 24;
        } else {
            if(mode==1) vector[0]=(i%2==0 ? 0.125F : -0.125F);
            if(!(*created)->Update(0,vector.data(),dim)) return 24;
        }
    }
    const double cpu_ms=1000.0*double(std::clock()-start)/CLOCKS_PER_SEC;
    const auto after=save(**created);
    std::cout<<std::setprecision(12)<<"{\"count\":2000,\"dim\":128,\"updates\":100,"
             <<"\"cpu_ms\":"<<cpu_ms<<",\"snapshot_unchanged\":"
             <<(before==after ? "true" : "false")<<",\"live_count\":"<<(*created)->Size();
    if(mode!=0) {
        std::cout<<",\"final_snapshot_hex\":\"";
        constexpr char hex[]="0123456789abcdef";
        for(unsigned char byte:after) std::cout<<hex[byte>>4]<<hex[byte&15];
        std::cout<<"\"";
    }
    std::cout<<"}\n";
    return 0;
}
int stress_faults(int mode=0) {
    constexpr uint64_t dim=17;
    auto created=vsag::lite::Index::Create(dim);
    if(!created) return 31;
    std::vector<float> vector(dim);
    for(int64_t row=0;row<32;++row) {
        for(uint64_t d=0;d<dim;++d) vector[d]=std::sin(float(row*7+int64_t(d))*0.17F);
        if(!(*created)->Add(row,vector.data(),dim)) return 32;
    }
    if(!(*created)->BuildGraph(vsag::lite::VectorStorage::RABITQ8,4,16)) return 33;
    const auto baseline=save(**created);
    int total=0;
    for(int64_t target : (mode==2 ? std::vector<int64_t>{0,5,20,31} : std::vector<int64_t>{0,5,20})) {
        for(uint64_t d=0;d<dim;++d) vector[d]=std::cos(float(d)*0.07F+float(target));
        vector[0]+=0.25F;
        bool completed=false;
        for(int point=0;point<512;++point) {
            std::istringstream input(baseline);
            auto loaded=vsag::lite::Index::Load(input);
            if(!loaded) return 34;
            failure_fired=false;
            allocations_after_failure=0;
            fail_after=point;
            bool ok;
            if(mode==2) {
                ok=(*loaded)->Remove(target);
                if(!ok && (!failure_fired || allocations_after_failure!=0)) return 38;
            } else {
                auto result=mode==1 ? (*loaded)->Add(1000+target,vector.data(),dim)
                                   : (*loaded)->Update(target,vector.data(),dim);
                ok=bool(result);
                if(!ok && result.error().type!=vsag::ErrorType::NO_ENOUGH_MEMORY) return 35;
            }
            fail_after=-1;
            failure_fired=false;
            if(ok) {completed=true;break;}
            ++total;
            if(save(**loaded)!=baseline) return 36;
        }
        if(!completed) return 37;
    }
    std::cout<<"PASS: "<<total<<(mode==2 ? " multi-row Remove failures retained exact snapshots; rollback allocations=0\n"
                       : (mode==1 ? " multi-row Add allocation failures retained exact snapshots\n"
                                  : " multi-row Update allocation failures retained exact snapshots\n"));
    return 0;
}
int main(int argc,char** argv) {
    if(argc>1 && std::string(argv[1])=="--stress-remove") return stress_faults(2);
    if(argc>1 && std::string(argv[1])=="--stress-add") return stress_faults(1);
    if(argc>1 && std::string(argv[1])=="--stress") return stress_faults();
    if(argc>1 && std::string(argv[1])=="--noop") return measure_update(0);
    if(argc>1 && std::string(argv[1])=="--remove") return measure_update(3);
    if(argc>1 && std::string(argv[1])=="--add") return measure_update(2);
    if(argc>1 && std::string(argv[1])=="--changed") return measure_update(1);
    auto created=vsag::lite::Index::Create(3);
    if(!created) return 1;
    auto& index=**created;
    float a[]={0,1,2}, b[]={2,1,0};
    if(!index.Add(42,a,3)||!index.Add(7,b,3)) return 2;
    auto built=index.BuildGraph(vsag::lite::VectorStorage::RABITQ8,2,8);
    if(argc>1) {
        if(built || built.error().type!=vsag::ErrorType::UNSUPPORTED_INDEX_OPERATION || index.Size()!=2 || index.ActiveBackend()!=vsag::lite::BackendKind::BRUTE_FORCE) return 3;
        std::cout<<"PASS: disabled build rejects RaBitQ and retains flat contents\n";
        return 0;
    }
    if(!built) return 4;
    const auto baseline=save(index);
    // Invalid-ID requests must not allocate a copied transaction before rejection.
    fail_after=0;
    auto duplicate=index.Add(42,a,3);
    const bool duplicate_no_allocation=fail_after==0;
    fail_after=-1;
    if(duplicate || duplicate.error().type!=vsag::ErrorType::INVALID_ARGUMENT || !duplicate_no_allocation) return 25;
    fail_after=0;
    auto missing=index.Update(999,a,3);
    const bool update_no_allocation=fail_after==0;
    fail_after=-1;
    if(missing || missing.error().type!=vsag::ErrorType::INVALID_ARGUMENT || !update_no_allocation) return 26;
    fail_after=0;
    const bool removed=index.Remove(999);
    const bool remove_no_allocation=fail_after==0;
    fail_after=-1;
    if(removed || !remove_no_allocation || save(index)!=baseline) return 27;
    int total_failures=0;
    for(int operation=0;operation<3;++operation) {
        int failures=0;
        bool success=false;
        for(int point=0;point<256;++point) {
            std::istringstream input(baseline);
            auto loaded=vsag::lite::Index::Load(input);
            if(!loaded) return 5;
            fail_after=point;
            bool ok;
            if(operation==2) ok=(*loaded)->Remove(42);
            else {
                auto result=operation==0 ? (*loaded)->Add(9,a,3) : (*loaded)->Update(42,b,3);
                ok=bool(result);
                if(!ok && result.error().type!=vsag::ErrorType::NO_ENOUGH_MEMORY) return 6;
            }
            fail_after=-1;
            if(ok) { success=true; break; }
            ++failures;
            if(save(**loaded)!=baseline) return 7;
        }
        if(!success || failures==0) return 8;
        total_failures+=failures;
    }
    std::cout<<"PASS: "<<total_failures<<" injected Add/Update/Remove allocation failures retained exact snapshots\n";
}
